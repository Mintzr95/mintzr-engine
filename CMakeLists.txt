cmake_minimum_required(VERSION 3.22)
project(MJUEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mju_core STATIC mju/core/application.cpp mju/core/engine.cpp mju/core/scene.cpp mju/io/scene_io.cpp mju/editor/editor.cpp mju/assets/asset_manager.cpp mju/tilemap/tilemap.cpp mju/ui/ui.cpp mju/physics/physics.cpp mju/particles/particles.cpp mju/resources/resources.cpp mju/scripting/script.cpp mju/diagnostics/debug.cpp mju/project/project.cpp)
target_include_directories(mju_core PUBLIC mju)
target_compile_features(mju_core PUBLIC cxx_std_20)
if(ANDROID)
 add_library(mju_android_native SHARED platform/android/android_entry.cpp mju/render/renderer_gles2.cpp)
 target_include_directories(mju_android_native PRIVATE mju)
 target_link_libraries(mju_android_native PRIVATE mju_core android log GLESv2)
endif()

option(MJU_BUILD_TESTS "Build MJU core tests" ON)
if(MJU_BUILD_TESTS)
 enable_testing()
 add_subdirectory(tests)
endif()
